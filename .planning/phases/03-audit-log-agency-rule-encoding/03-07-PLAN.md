# Plan 03-07: Cascade trigger integration test + cascade_review_queue cross-tenant pen tests

## Objective

Phase 3 SC#5 is the gate this plan delivers:
> "A new `agency_rule_version` insert fires a trigger that enqueues a `cascade.review` job for every `program_version` referencing the now-superseded agency version (verified via integration test)."

The Wave 0 plan (03-01) shipped the `enqueue_agency_cascade()` plpgsql trigger function, the `CREATE TRIGGER agency_rule_version_cascade` AFTER INSERT, and the `cascade_review_queue` table with FORCE RLS + two-policy shape. This plan ships:

1. **D-17 integration test** at `tests/cascade/cascade-trigger.test.ts` — seeds 2 tenants × 1 program_version each FK'd to the same prior `agency_rule_version`, then INSERTs a new `agency_rule_version` using the two-step convention, then asserts the trigger created exactly 2 `cascade_review_queue` rows (one per tenant). PLUS the Pitfall PG-4 short-circuit test (initial INSERT with `superseded_by IS NULL` produces zero queue rows).
2. **Cross-tenant pen tests** at `tests/rls/cascade-review-queue-cross-tenant.test.ts` — adapts the Phase 2 D-19 6-case matrix template to the new tenant-scoped `cascade_review_queue` table (closes AGY-08 cross-tenant RLS coverage matching Phase 1 D-03 + Phase 2 D-19).
3. **Helper extension** at `tests/rls/seedTwoTenants.ts::seedTwoTenantsWithProgramVersions` — variant of the existing `seedTwoTenants` that creates ONE program_version per tenant pointing at the SAME prior `agency_rule_version` but skips the program_rule INSERT (cascade trigger only needs program_version, not program_rule per Pitfall PG-9 — the trigger JOINs program_version ONLY, not agency_rule).

This plan depends on Plan 03-02 (FNMA seed) because the cascade trigger test needs at least one agency_rule_version with `version_label='FNMA-SEL-2026-04'` to use as the "prior" version. Plan 03-01 alone is insufficient because the Wave 0 USDA stub has zero child rules and the test fixture works under FNMA.

     * Cascade trigger integration test (Phase 3 SC#5 / D-17 / AGY-08).
     *
     * Verifies the enqueue_agency_cascade() trigger (created in Plan 03-01 Task 4)
     * fires AFTER INSERT ON agency_rule_version and writes one cascade_review_queue
     * row per affected program_version.
     *
     * Test cases:
     *   1. Two-step convention with 2 tenants × 1 program_version each → 2 queue rows
     *   2. Pitfall PG-4: initial INSERT (superseded_by IS NULL) → 0 queue rows
     *   3. Pitfall PG-9: multiple agency_rule under prior → still N program_version
     *      rows, not N × ruleCount (the trigger JOINs program_version ONLY)
     */
    import { describe, expect, it } from 'vitest';
    import { seedTwoTenantsWithProgramVersions } from '../rls/seedTwoTenants.js';

    describe('cascade trigger (Phase 3 SC#5 / AGY-08)', () => {
      it('inserts one cascade_review_queue row per affected program_version across tenants (D-17)', async () => {
        const adminPool = globalThis.__pgAdminPool;

        // Setup: get the FNMA prior agency_rule_version (already seeded via pnpm db:seed).
        const adminClient = await adminPool.connect();
        let priorArvId: string;
        try {
          const { rows } = await adminClient.query<{ id: string }>(
            `SELECT id::text FROM agency_rule_version
             WHERE agency = 'FNMA' AND version_label = 'FNMA-SEL-2026-04' LIMIT 1`,
          );
          if (!rows[0]) throw new Error('FNMA-SEL-2026-04 not seeded; expected from pnpm db:seed');
          priorArvId = rows[0].id;
        } finally {
          adminClient.release();
        }

        // Seed 2 tenants × 1 program_version each → both FK to the FNMA prior.
        const setup = await seedTwoTenantsWithProgramVersions(
          globalThis.__pgPool,
          adminPool,
          priorArvId,
        );

        // Run two-step convention. Pre-allocate the new ARV uuid so we can use
        // it both in UPDATE prior.superseded_by and INSERT new.id.
        const fireClient = await adminPool.connect();
        try {
          await fireClient.query('BEGIN');

          const newIdRow = await fireClient.query<{ id: string }>(
            `SELECT gen_random_uuid()::text AS id`,
          );
          const newArvId = newIdRow.rows[0]!.id;

          // STEP 1: UPDATE prior — set superseded_by + close effective_period.
          // Use a year far from realistic data to avoid colliding with
          // agency_rule_version_no_overlap EXCLUDE.
          const closeYear = 8000000 + Math.floor(Math.random() * 100000);
          await fireClient.query(
            `UPDATE agency_rule_version
               SET superseded_by = $1,
                   effective_period = daterange(lower(effective_period), $2::date, '[)')
             WHERE id = $3`,
            [newArvId, `${closeYear}-04-01`, priorArvId],
          );

          // STEP 2: INSERT new agency_rule_version. Trigger fires AFTER INSERT.
          await fireClient.query(
            `INSERT INTO agency_rule_version (id, agency, version_label, source_url, effective_period)
             VALUES ($1, 'FNMA', 'FNMA-SEL-CASCADE-TEST-' || gen_random_uuid()::text,
                     'https://selling-guide.fanniemae.com/cascade-test', $2::daterange)`,
            [newArvId, `[${closeYear}-04-01,infinity)`],
          );

          // Assert: exactly 2 queue rows, one per tenant.
          const { rows: queue } = await fireClient.query<{
            tenant_id: string;
            program_version_id: string;
            prior_agency_rule_version_id: string;
            new_agency_rule_version_id: string;
            status: string;
          }>(
            `SELECT tenant_id::text AS tenant_id,
                    program_version_id::text AS program_version_id,
                    prior_agency_rule_version_id::text AS prior_agency_rule_version_id,
                    new_agency_rule_version_id::text AS new_agency_rule_version_id,
                    status
             FROM cascade_review_queue
             WHERE new_agency_rule_version_id = $1
             ORDER BY tenant_id`,
            [newArvId],
          );
          expect(queue).toHaveLength(2);
          for (const row of queue) {
            expect(row.status).toBe('pending');
            expect(row.new_agency_rule_version_id).toBe(newArvId);
            expect(row.prior_agency_rule_version_id).toBe(priorArvId);
          }
          // Tenants A and B both represented.
          const tenantIds = queue.map((r) => r.tenant_id).sort();
          expect(tenantIds).toEqual([setup.tenantA, setup.tenantB].sort());
          // Program versions A and B both represented.
          const programVersionIds = queue.map((r) => r.program_version_id).sort();
          expect(programVersionIds).toEqual(
            [setup.programVersionA, setup.programVersionB].sort(),
          );

          await fireClient.query('ROLLBACK');
        } finally {
          fireClient.release();
        }
      });

      it('initial agency version (no prior) inserts zero queue rows (Pitfall PG-4)', async () => {
        const adminPool = globalThis.__pgAdminPool;
        const fireClient = await adminPool.connect();
        try {
          await fireClient.query('BEGIN');

          // Insert a fresh agency_rule_version with no superseded_by relationship.

             RETURNING id::text`,

          );
          const newArvId = inserted[0]!.id;

          // Assert: cascade_review_queue is unchanged (zero rows for this new ARV).
          const { rows: queue } = await fireClient.query(
            `SELECT 1 FROM cascade_review_queue WHERE new_agency_rule_version_id = $1`,
            [newArvId],
          );
          expect(queue).toHaveLength(0);

          await fireClient.query('ROLLBACK');
        } finally {
          fireClient.release();
        }
      });

      it('multiple agency_rule under prior → trigger still produces N queue rows, not N × ruleCount (Pitfall PG-9)', async () => {
        const adminPool = globalThis.__pgAdminPool;

        // Setup: get FNMA prior (which already has 8 child agency_rule rows from the seed).

        let agencyRuleCount: number;

             WHERE agency = 'FNMA' AND version_label = 'FNMA-SEL-2026-04' LIMIT 1`,
          );
          priorArvId = arvRow.rows[0]!.id;

          const ruleRow = await adminClient.query<{ count: string }>(
            `SELECT count(*)::text AS count FROM agency_rule
             WHERE agency_rule_version_id = $1`,
            [priorArvId],
          );
          agencyRuleCount = parseInt(ruleRow.rows[0]!.count, 10);
          // Plan 03-02 ships 8 FNMA agency_rule rows; sanity-check we're testing
          // against the post-seed state.

        } finally {

        }

        // Seed 2 tenants × 1 program_version each FK'd to the FNMA prior.

          globalThis.__pgPool,
          adminPool,
          priorArvId,

        const fireClient = await adminPool.connect();
