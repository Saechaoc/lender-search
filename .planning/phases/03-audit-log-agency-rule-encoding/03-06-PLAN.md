    ```markdown
    Annual FHFA per-county loan limit data, committed as CSV and loaded by
    `pnpm db:seed` via `scripts/seed-agency.ts::seedFhfaYear`.

    ## File: `<year>-conforming-limit-values-by-county.csv`

    Source: https://www.fhfa.gov/data/conforming-loan-limit

    Each year FHFA publishes two variants:
    - **FINAL FLAT** — actual published per-county limit (what lenders use). **WE COMMIT THIS.**
    - **HERA-BASED** — statutory derivation; informational only.

    Columns:
    - FIPS State Code (text, 2 chars, leading zeros preserved)
    - FIPS County Code (text, 3 chars, leading zeros preserved)
    - County Name (quoted-field-safe; e.g. "Saint Mary's County")
    - State (2-char alpha code)
    - CBSA Number (5-char numeric or empty)
    - One/Two/Three/Four-Unit Limit (integer dollars)

    The loader composes `county_fips = fips_state + fips_county` (text(5) preserving zeros).

    ## Annual update procedure (D-23)

    1. Download the new year's FINAL FLAT CSV from FHFA.
    2. Commit the file as `lib/agency-seeds/fhfa/<year>-conforming-limit-values-by-county.csv`.
    3. Update `seedFhfaYear` invocation in `scripts/seed-agency.ts main()` with the new year.
    4. Run `pnpm db:seed`. The loader's two-step daterange close convention:
       - UPDATE the previous infinity-bounded version's effective_period to `[<prev_year>-01-01, <new_year>-01-01)`.
       - INSERT the new year's row with `[<new_year>-01-01, infinity)`.
       - INSERT the ~3,200 conforming_loan_limit_county child rows.
    5. The conforming_loan_limit_version EXCLUDE constraint allows two years
       to coexist as long as their effective_periods don't overlap (the close
       step prevents overlap).

    ## High-cost derivation (Open Question 7)

    `is_high_cost` is stored at load time per Open Question 7. The loader reads the
    one-unit baseline from FHFA's annual announcement (2026 = $832,750) and
    sets `is_high_cost = (one_unit_baseline > national_baseline)`. Per-year
    threshold passed as a constant in the seed call so historical loaders
    stay accurate.
    ```

    ```typescript
    import { parse } from 'csv-parse/sync';
    import { readFile } from 'node:fs/promises';
    import { join } from 'node:path';
    import { z } from 'zod';

    // FHFA 2026 baseline per Assumption A5 — the threshold above which is_high_cost=true.
    const FHFA_2026_ONE_UNIT_BASELINE = 832750;

    const fhfaRowSchema = z.object({
      'FIPS State Code': z.string().regex(/^\d{1,2}$/).transform((s) => s.padStart(2, '0')),
      'FIPS County Code': z.string().regex(/^\d{1,3}$/).transform((s) => s.padStart(3, '0')),
      'County Name': z.string().min(1).max(120),
      'State': z.string().length(2),
      'CBSA Number': z.string().regex(/^\d{0,5}$/).optional().nullable(),
      'One-Unit Limit': z.coerce.number().int().positive(),
      'Two-Unit Limit': z.coerce.number().int().positive().optional().nullable(),
      'Three-Unit Limit': z.coerce.number().int().positive().optional().nullable(),
      'Four-Unit Limit': z.coerce.number().int().positive().optional().nullable(),
    });

    /**
     * seedFhfaYear — extends the Plan 03-01 skeleton with the real CSV-parse + Zod-validate
     *   + batch-INSERT path. Per D-23 two-step daterange close convention. Per Pitfall PG-7,
     *   csv-parse handles BOM + quoted fields. Per Open Question 7, is_high_cost stored at
     *   load time using the per-year baseline (FHFA 2026 = $832,750).
     */
    export async function seedFhfaYear(year: number, csvPath: string): Promise<void> {
      const baseline = year === 2026 ? FHFA_2026_ONE_UNIT_BASELINE
        : null; // Future years: extend the constant table when adding.
      if (baseline === null) {
        throw new Error(`seedFhfaYear: no baseline configured for year ${year} (Open Question 7)`);
      }

      const csvBytes = await readFile(csvPath);
      // csv-parse 5.6 sync API: parse returns array; columns:true gives row-as-object.
      // bom:true strips UTF-8 BOM (Pitfall PG-7); trim:true cleans cell whitespace;
      // skip_empty_lines:true ignores trailing blank lines.
      const records = parse(csvBytes, {
        columns: true,
        bom: true,
        trim: true,
        skip_empty_lines: true,
      }) as unknown[];

      const client = await pool.connect();
      try {
        await client.query('BEGIN');

        // Two-step daterange close convention (D-23).
        await client.query(
          `UPDATE conforming_loan_limit_version
             SET effective_period = daterange(lower(effective_period), $1::date, '[)')
           WHERE upper(effective_period::text) = 'infinity'
             AND year < $2`,
          [`${year}-01-01`, year],
        );

        // Get-or-create version row keyed on year.
        let versionId: string;
        const existing = await client.query<{ id: string }>(
          `SELECT id::text FROM conforming_loan_limit_version WHERE year = $1 LIMIT 1`,
          [year],
        );
        if (existing.rows[0]) {
          versionId = existing.rows[0].id;
        } else {
          const created = await client.query<{ id: string }>(
            `INSERT INTO conforming_loan_limit_version (year, effective_period, source_url)
             VALUES ($1, $2::daterange, $3)
             RETURNING id::text`,
            [
              year,
              `[${year}-01-01,infinity)`,
              'https://www.fhfa.gov/data/conforming-loan-limit',
            ],
          );
          versionId = created.rows[0]!.id;
        }

        // Per-row Zod validation + INSERT. ~3,200 rows; ~10ms total Zod overhead.
        // Idempotency via INSERT ... ON CONFLICT DO NOTHING on composite PK
        // (limit_version_id, county_fips). Re-running the loader is a no-op when state is current.
        for (const raw of records) {
          const row = fhfaRowSchema.parse(raw);
          const countyFips = `${row['FIPS State Code']}${row['FIPS County Code']}`;
          const oneUnit = row['One-Unit Limit'];
          const isHighCost = oneUnit > baseline;

          await client.query(
            `INSERT INTO conforming_loan_limit_county (
               limit_version_id, county_fips, state_code,
               one_unit_baseline, two_unit_baseline, three_unit_baseline, four_unit_baseline,
               is_high_cost
             )
             VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
             ON CONFLICT (limit_version_id, county_fips) DO NOTHING`,
            [
              versionId, countyFips, row['State'],
              oneUnit,
              row['Two-Unit Limit'] ?? null,
              row['Three-Unit Limit'] ?? null,
              row['Four-Unit Limit'] ?? null,
              isHighCost,
            ],
          );
        }

        await client.query('COMMIT');
      } catch (err) {
        try { await client.query('ROLLBACK'); } catch { /* swallow */ }
        throw err;
      } finally {
        client.release();
      }
    }
    ```
