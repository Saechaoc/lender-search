    /**
     * Vitest setupFiles for tests/schema/ — exposes __pgPool (app_user) and
     * __pgAdminPool (postgres) globals for fixture helpers.
     *
     * Per Pitfall G: agency-rule INSERTs (D-20.6 + D-20.7) require the postgres
     * superuser connection because agency_rule's system_write policy uses
     * `to: system_role` and only postgres has system_role GRANTed. App_user
     * (NOBYPASSRLS NOSUPERUSER) cannot write to agency_rule.
     *
     * Plan 02-07 [BLOCKING] migrate already applied all migrations; this file
     * does NOT run drizzle-kit migrate. It only opens connections and exposes
     * them on globalThis.
     *
     * Phase 1's vitest.config.ts setupFiles (./tests/rls/setup.ts) does FORCE-RLS
     * sanity checks — Plan 02-08 does NOT need those because schema-constraint
     * tests connect with the appropriate role per test, and FORCE was already
     * verified by Plan 02-07's introspection.
     */
    import { config as loadDotenv } from 'dotenv';
    import { Pool } from 'pg';
    import { afterAll, beforeAll } from 'vitest';

    loadDotenv({ path: '.env.local' });
    loadDotenv({ path: '.env' });

    declare global {
      // eslint-disable-next-line no-var
      var __pgPool: Pool;
      // eslint-disable-next-line no-var
      var __pgAdminPool: Pool;
    }

    if (!process.env.DATABASE_URL) {
      throw new Error('DATABASE_URL must be set for tests/schema/ to run.');
    }

    // DATABASE_MIGRATION_URL is the postgres superuser connection. Falls back to
    // a derived URL if only DATABASE_URL is available (mirroring tests/rls/
    // global-setup.ts pattern).
    const ADMIN_URL =
      process.env.DATABASE_MIGRATION_URL ??
      process.env.DATABASE_URL?.replace('app_user:app_user_password', 'postgres:postgres') ??
      '';

    if (!ADMIN_URL) {
      throw new Error('DATABASE_MIGRATION_URL or DATABASE_URL must yield a postgres-superuser URL for agency-table writes (Pitfall G).');
    }

    beforeAll(async () => {
      globalThis.__pgPool = new Pool({
        connectionString: process.env.DATABASE_URL,
        max: 5,
      });
      globalThis.__pgAdminPool = new Pool({
        connectionString: ADMIN_URL,
        max: 3,
      });
    });

    afterAll(async () => {
      await globalThis.__pgPool?.end();
      await globalThis.__pgAdminPool?.end();
    });

    /**
     * tests/schema/fixtures/seed.ts — bootstrap helpers for schema-constraint
     * tests. Reuses the Phase 1 Pitfall 7 bootstrap pattern (set_config +
     * INSERT tenant with id=GUC) for tenant-scoped writes; uses postgres

     */
    import type { Pool } from 'pg';
    import type { DerogSeasoning } from '../../../lib/rules/schemas/derog-seasoning.js';

    export interface TenantSeed {
      tenantId: string;
      programId: string;
      citationId: string;
      programVersionId: string;
    }

    /**
     * Seed a tenant + program + citation + draft program_version under app_user.
     * Returns UUIDs for use in tests.
     *
     * The bootstrap pattern (Pitfall 7): generate uuid → set_config app.tenant_id
     * → INSERT tenant (id matches GUC) → INSERT subsequent rows under the same GUC.
     */
    export async function seedTenantWithProgramAndCitation(
      pool: Pool,
      adminPool: Pool,
      agencyRuleVersionId: string,
    ): Promise<TenantSeed> {
      const client = await pool.connect();
      try {
        await client.query('BEGIN');

        const idResult = await client.query<{ id: string }>(`SELECT gen_random_uuid()::text AS id`);
        const tenantId = idResult.rows[0]!.id;

        // Set GUC to the new tenant's id BEFORE the INSERT so WITH CHECK passes.
        await client.query(`SELECT set_config('app.tenant_id', $1, true)`, [tenantId]);

        await client.query(
          `INSERT INTO tenant (id, kind, name) VALUES ($1, 'BROKERAGE', 'Schema Test Tenant')`,
          [tenantId],
        );

        const programResult = await client.query<{ id: string }>(
          `INSERT INTO program (tenant_id, lender, channel, name)
           VALUES ($1, 'PennyMac TPO', 'wholesale', 'Bank Statement 24-mo DSCR')
           RETURNING id::text`,
          [tenantId],
        );
        const programId = programResult.rows[0]!.id;

        const citationResult = await client.query<{ id: string }>(
          `INSERT INTO rule_citation (tenant_id, source_url, excerpt)
           VALUES ($1, 'https://lender.example/matrix.pdf', 'Test excerpt for schema fixture')
           RETURNING id::text`,
          [tenantId],
        );
        const citationId = citationResult.rows[0]!.id;

        // program_version with state='draft' (no overlap risk yet).
        const versionResult = await client.query<{ id: string }>(
          `INSERT INTO program_version (
             tenant_id, program_id, agency_rule_version_id,
             effective_period, state, source_document_fingerprint
           ) VALUES ($1, $2, $3, '[2026-01-01,2027-01-01)', 'draft', 'sha256:fixture')
           RETURNING id::text`,
          [tenantId, programId, agencyRuleVersionId],
        );
        const programVersionId = versionResult.rows[0]!.id;

        await client.query('COMMIT');
        return { tenantId, programId, citationId, programVersionId };
      } catch (err) {
        try {
          await client.query('ROLLBACK');
        } catch {
          /* swallow */
        }
        throw err;
      } finally {
        client.release();
      }
    }

     * the postgres superuser (Pitfall G — agency_rule's system_write policy
     * requires system_role; only postgres has it GRANTed). Citation lives in

      effectivePeriod: string,

      const client = await adminPool.connect();
      try {
        await client.query('BEGIN');

        // Create a system-tenant fixture row first (agency citations are tenant-
        // scoped per current schema; Phase 3 may revise to dedicate a SYSTEM

        const systemTenantResult = await client.query<{ id: string }>(
          `INSERT INTO tenant (id, kind, name)
           VALUES (gen_random_uuid(), 'SYSTEM', 'Agency Hand-Authoring System Tenant')
           ON CONFLICT DO NOTHING
           RETURNING id::text`,
        );
        let systemTenantId: string;
        if (systemTenantResult.rows[0]) {
          systemTenantId = systemTenantResult.rows[0].id;
        } else {
          const lookup = await client.query<{ id: string }>(
            `SELECT id::text FROM tenant WHERE kind = 'SYSTEM' LIMIT 1`,
          );
          systemTenantId = lookup.rows[0]!.id;
        }

        // Set GUC so RLS policies on tenant + rule_citation pass for the system tenant.

        const citationResult = await client.query<{ id: string }>(
